function computeAndRenderTeams() { const teamCount = parseInt(document.getElementById('teamCount').value) || 4; const mode = publishedTeamSettings.mode || 'current'; const monthFilter = publishedTeamSettings.monthFilter || 'all'; // 각 선수의 기준 점수에 현재 입력된 1게임 점수(또는 실시간 점수)를 동적으로 반영 const getPlayerBaseSortingScore = (p) => { let score = p.score; if (mode === 'history') { let tot = 0, cnt = 0; history.forEach(h => { if (monthFilter !== 'all' && !h.date.replace(/[-/]/g, '-').startsWith(monthFilter)) return; let rec = h.records.find(r => r.name === p.name); if (rec && parseFloat(rec.avg) > 0) { tot += parseFloat(rec.avg); cnt++; } }); if (cnt > 0) score = parseFloat((tot / cnt).toFixed(1)); } // 실시간 입력된 1게임 점수가 있다면 이를 즉시 기준 점수에 합산 혹은 반영하여 팀 재편성에 활용 let pLive = liveScores[p.id] || {}; let liveG1 = parseFloat(pLive.g1); if (!isNaN(liveG1) && liveG1 > 0) { // 예: 당일 점수 기준일 때 실시간 입력된 1게임 점수를 반영하고 싶다면 아래와 같이 조율 가능합니다. score = (score + liveG1) / 2; // 또는 liveG1 자체를 가중 반영 } return score; }; let evalPlayers = matchResults.map(p => ({ ...p, sortingScore: getPlayerBaseSortingScore(p) })); evalPlayers.sort((a, b) => b.sortingScore - a.sortingScore); let teams = Array.from({ length: teamCount }, () => ({ memberIds: new Set(), members: [], totalScore: 0, liveSumScore: 0 })); evalPlayers.forEach((player, index) => { let tIdx = Math.floor(index / teamCount) % 2 === 0 ? index % teamCount : teamCount - 1 - (index % teamCount); if (Math.floor(index / teamCount) >= 1) { teams.sort((a, b) => a.totalScore - b.totalScore); tIdx = teams.indexOf(teams[0]); } let pLive = liveScores[player.id] || {}; let dispScore = parseFloat(pLive.g1) || 0; teams[tIdx].memberIds.add(player.id); teams[tIdx].members.push({ ...player, displayGameScore: dispScore }); teams[tIdx].totalScore += player.sortingScore; teams[tIdx].liveSumScore += dispScore; }); const uniqueLiveScores = [...new Set(teams.map(t => t.liveSumScore))].sort((a, b) => b - a); document.getElementById('teamsContainer').innerHTML = teams.map((team, idx) => { const avg = team.members.length > 0 ? (team.totalScore / team.members.length).toFixed(1) : 0; let rankClass = ''; if (team.liveSumScore > 0) { if (team.liveSumScore === uniqueLiveScores[0]) rankClass = 'rank-1'; else if (team.liveSumScore === uniqueLiveScores[1]) rankClass = 'rank-2'; else rankClass = 'rank-3'; } return `

팀 ${idx + 1} (1G 합산: ${team.liveSumScore})

기준총점: ${team.totalScore.toFixed(0)} | 평균: ${avg}
${team.members.map(m => `
${m.name} (기준:${m.sortingScore.toFixed(1)}) ${m.displayGameScore > 0 ? m.displayGameScore + '점' : '-'}
`).join('')}
`; }).join(''); }
